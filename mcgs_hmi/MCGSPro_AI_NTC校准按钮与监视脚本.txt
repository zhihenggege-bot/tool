                n316CalReadback = R2584_4_316_NTC2盘管2偏移
            EndIf
            If n316CalWriteChannel = 3 Then
                n316CalReadback = R2585_4_316_NTC3排气1偏移
            EndIf
            If n316CalWriteChannel = 4 Then
                n316CalReadback = R2586_4_316_NTC4排气2偏移
            EndIf
            If n316CalWriteChannel = 5 Then
                n316CalReadback = R2587_4_316_NTC5吸气1偏移
            EndIf
            If n316CalWriteChannel = 6 Then
                n316CalReadback = R2588_4_316_NTC6吸气2偏移
            EndIf
            If n316CalWriteChannel = 7 Then
                n316CalReadback = R2589_4_316_NTC7阀后1偏移
            EndIf
        EndIf
    EndIf

    If n316CalWriteStatus = 3 Then
        n316CalWriteCheck = 0
        n316CalResult = 106
    Else
        If n316CalWriteStatus = 2 Then
            If n316CalSawBusy = 1 Then
                If n316CalReadback = n316CalExpectedOffset Then
                    n316CalWriteCheck = 0
                    n316CalSaveWait = 1
                    n316CalResult = 4
                Else
                    n316CalWriteCheck = 0
                    n316CalResult = 107
                EndIf
            EndIf
        EndIf
    EndIf

    If n316CalTimeout >= 100 Then
        n316CalWriteCheck = 0
        n316CalResult = 108
    EndIf
EndIf

If n316CalSaveWait > 0 Then
    n316CalSaveWait = n316CalSaveWait + 1
    If n316CalSaveWait >= 25 Then
        n316CalSaveWait = 0
        n316CalResult = 5
    EndIf
EndIf
