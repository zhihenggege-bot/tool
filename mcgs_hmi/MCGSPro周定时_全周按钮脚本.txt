'========================================================
'二、“启用周定时”按钮的抬起脚本
'========================================================
If nScheduleClockError <> 0 Then
    nScheduleResult = 101
Else
    If nScheduleConfigError <> 0 Then
        nScheduleResult = 102
    Else
        bScheduleMasterEnable = 1
        nScheduleExitStep = 0
        nScheduleWriteCooldown = 0
        nScheduleResult = 1
        nScheduleModeOne = 1
        R0165_定时模式使能 = 1
    EndIf
EndIf
