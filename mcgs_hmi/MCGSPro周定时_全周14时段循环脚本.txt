        nScheduleClockError = 0
        nNowWeekMin = (nRTCWeek - 1) * 1440 + nRTCHour * 60 + nRTCMinute

        nScheduleResult = 101
    EndIf

'--------------------------------------------------------
'6. 停机并退出定时
'先写1388=0，等709=0，再写165=0
'--------------------------------------------------------

    nScheduleTarget = 0

        If nScheduleWriteCooldown = 0 Then
            R1388_定时运行目标 = 0
            nScheduleWriteCooldown = 5

        EndIf
    Else
        If R0709_开关机步骤 = 0 Then
            If R0165_定时模式使能 <> 0 Then
                If nScheduleWriteCooldown = 0 Then
                    R0165_定时模式使能 = 0
                    nScheduleWriteCooldown = 5
                    nScheduleResult = 8
                EndIf
            Else
                nScheduleExitStep = 0
                nScheduleResult = 5
            EndIf
        EndIf
    EndIf
Else
    '----------------------------------------------------
    '7. 直接退出定时并保持当前状态
    '只退出165，不主动改1388
    '退出后163选择的本地/远程/PLC来源立即接管

        If R0165_定时模式使能 <> 0 Then
            If nScheduleWriteCooldown = 0 Then
                R0165_定时模式使能 = 0
                nScheduleWriteCooldown = 5
                nScheduleResult = 9
            EndIf
        Else
            nScheduleExitStep = 0
            nScheduleResult = 6
        EndIf
    Else

        '8. 正常周定时控制
        '时钟或设置错误时保持PSC330当前目标，不发送新命令

            If nScheduleClockError = 0 Then
                If nScheduleConfigError = 0 Then
                    '必须先确认165精确读回1

                            R0165_定时模式使能 = 1
                            nScheduleWriteCooldown = 5
                            nScheduleResult = 1
                        EndIf

                        '165已为1后才写1388

                            If nScheduleWriteCooldown = 0 Then
                                R1388_定时运行目标 = nScheduleTarget
                                nScheduleWriteCooldown = 5
                                nScheduleResult = 3
                            EndIf
                        Else

                                nScheduleResult = 2
                            Else
                                nScheduleResult = 4
                            EndIf
                        EndIf
                    EndIf
                EndIf
            EndIf
        EndIf
